🐍 Typed Async Python SDK

Support for Web (batchexecute) and Android (gRPC) transports.

import asyncio
from notebooklm import NotebookLMClient

async def main():
    # Supports Web and Android backends with automatic session healing
    async with NotebookLMClient.from_storage(backend="web") as client:
        nb = await client.notebooks.create("Quantum Research 2026")
        await client.sources.add_url(nb.id, "https://arxiv.org/abs/2603.09182", wait=True)
        res = await client.chat.ask(nb.id, "Summarize key gate fidelities")
        print("Answer:", res.answer)

asyncio.run(main())
